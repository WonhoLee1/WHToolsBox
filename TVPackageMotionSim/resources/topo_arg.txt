# WHT Topography Optimization — 기본 옵션 설정 파일
# 사용법: python wht_topo/run_topo.py @options.txt
# 주석(#)으로 시작하는 줄은 무시됩니다.
# 비활성 옵션은 앞의 # 를 제거하면 활성화됩니다.

# ── 섀시 형상 (mm) ──────────────────────────────────────────────────────────
# 섀시 외형 치수와 메시 크기를 지정합니다.
# tray-width/length/height: 섀시 폭·길이·높이 (플랜지 포함 외곽 치수)
# mesh-xy: 바닥·측벽 평면 방향 메시 크기. 작을수록 정밀하지만 해석 시간 증가
# mesh-z: 측벽 높이 방향 메시 크기
# draft-angle: 측벽 기울기 각도 (deg). 0이면 수직 측벽, 양수이면 외향 경사
--tray-width    1800.0
--tray-length   1200.0
--tray-height   40.0
--mesh-xy       30.0
--mesh-z        10.0
--draft-angle   25.0

# ── 재료 물성 ────────────────────────────────────────────────────────────────
# 셸 요소에 적용되는 등방성 탄성 재료 물성입니다.
# mat-E  : 탄성계수 [MPa]. 강재=210000, 알루미늄=70000
# mat-nu : 포아송비. 강재≈0.3, 알루미늄≈0.33
# mat-rho: 밀도 [tonne/mm³]. 강재=7.85e-9, 알루미늄=2.7e-9
#          (단위계: tonne·mm·s → 힘 N, 질량 tonne)
# mat-t  : 기본 판 두께 [mm]
--mat-E         210000.0
--mat-nu        0.3
--mat-rho       7.85e-09
--mat-t         0.6

# ── 최적화 기본 설정 ─────────────────────────────────────────────────────────
# MMA(Method of Moving Asymptotes) 기반 토포그래피 최적화 제어 파라미터입니다.
# iters      : 최대 반복 횟수. 수렴 전 강제 종료 기준
# bead-height: 비드 최대 돌출 높이 [mm]
# min-width  : 비드 최소 폭 [mm]. 제조 가능성 제약 (너무 좁은 비드 방지)
# bead-area  : 비드가 차지할 수 있는 바닥 면적 비율 (0~1). 0.30 = 전체의 30%
--iters         15
--bead-height   10.0
--min-width     60.0
--bead-area     0.30

# ── 목적함수 옵션 ────────────────────────────────────────────────────────────
# 다중 하중 케이스를 하나의 스칼라 목적함수로 집약하는 방식을 설정합니다.
# normalize-obj: 각 케이스를 초기 컴플라이언스로 나눠 정규화.
#               하중 크기가 케이스마다 다를 때 균등 반영을 위해 권장
# obj-type     : sum=가중합, max=softmax 최악케이스, sum+max=두 방식 0.5씩 혼합
# obj-alpha    : softmax 온도 파라미터. 값이 클수록 최악 케이스에 집중
# freq-penalty : 고유진동수 패널티 [W F0_HZ].
#               W=패널티 가중치, F0=목표 진동수(Hz). f1 < F0 이면 패널티 부과
# --normalize-obj
--obj-type      max
--obj-alpha     10.0
--freq-penalty  0.0  0.0

# ── 병렬 해석 옵션 ───────────────────────────────────────────────────────────
# 하중 케이스를 멀티스레드로 병렬 해석합니다.
# n-workers: 병렬 스레드 수. 케이스 수보다 크게 설정해도 케이스 수로 자동 제한.
#            BLAS 내부 멀티스레드와 경합을 피하기 위해 CPU 코어 수의 절반 권장.
#            예: 8코어 CPU → --n-workers 4
--n-workers     4

# ── 비드 형상 및 제조 제약 ───────────────────────────────────────────────────
# 비드 형상의 대칭, 연결성, 방향, 배제 영역 등 제조 제약을 설정합니다.
# sym-x       : X축 기준 좌우 대칭 강제. 대칭 섀시에 권장 (--no-sym-x 로 해제)
# bead-connect: 고립된 비드를 인접 비드에 자동 연결 (--no-bead-connect 로 해제)
# connect-gap : 비드 자동 연결 시 허용 최대 갭 [mm]
# height-steps: 비드 높이 이산화 단계 수. 1=연속, 2={0, h_max}, 3={0, h/2, h_max}
# draw-dir    : 비드 돌출 방향 벡터 (쉼표 구분). 0,0,-1=아래, 0,0,1=위
# exclude-rect: 비드 배제 사각형 영역 [중심X,중심Y,폭W,높이H] (mm).
#              여러 영역 지정 시 옵션을 반복: --exclude-rect ... --exclude-rect ...
# exclude-poly: 비드 배제 다각형 영역 [X1,Y1,X2,Y2,...] (mm)
--sym-x
--bead-connect
--connect-gap   120.0
--height-steps  1
--draw-dir      0,0,-1
# --exclude-rect  CX,CY,W,H
# --exclude-poly  X1,Y1,X2,Y2,...

# ── 동적 ESL 통합 최적화 (모드 C/D) ─────────────────────────────────────────
# 실측 CSV 위치 데이터로부터 등가 정하중(ESL)을 추출하여 최적화에 반영합니다.
# dynamic-opts : CSV 파일 경로와 분석 구간 지정 (경로,시작s,종료s).
#               복수 시나리오: --dynamic-opts rear.csv,0.4,0.6 bump.c.5,1.8
# add-inertia  : 코너 가속도 쌍선형 보간으로 관성 하중(F=-ma)을 전 노드에 인가
# n-top        : 에너지 기준 상위 N개 스냅샷을 ESL로 추출
# n-windows    : 시간 이력을 N개 구간으로 분할하여 스냅샷 추출
# use-global-z : 로컬 Z 대신 글로벌 Z 궤적을 직접 사용
# w-dynamic    : ESL 스냅샷 가중치. CSV별 복수 지정: --w-dynamic 1.0 0.8 0.8
# w-peak       : 요소별 변형에너지 피크 ESL 가중치 (0.0=비활성)
# iterative-esl: 매 이터레이션마다 현재 형상 기준으로 ESL 재추출 (기본 활성)
#               --no-iterative-esl: 최적화 전 1회만 추출 후 고정
# no-static    : 정적 하중 케이스 전체 비활성 (동적 ESL만 사용)
# --dynamic-opts  CSV경로,T_START,T_END
--add-inertia
--n-top         5
--n-windows     10
# --use-global-z
--w-dynamic     1.0
--w-peak        0.5
--iterative-esl
# --no-iterative-esl
# --no-static
--dt            0.001
--esl-skip-tol 0.05
--parallel-scenarios
--modal-dynamic

# ── 정적 하중 케이스 설정 ────────────────────────────────────────────────────
# 각 정적 하중 케이스의 목적함수 가중치(W)와 적용 하중 크기(F_N)를 설정합니다.
# 형식: --w-xxx  W  F_N  (W=목적함수 가중치, F_N=하중 [N], 음수=하향)
#
# w-basic-weight: 자중 기반 하중 자동계산 스케일.
#   0     : F_N 입력값을 그대로 적용
#   1.0   : 루프형 질량 행렬로 산출한 총 자중(W=M*9806 N)을 F_N으로 대체
#   기타값: W_chassis * SCALE 을 F_N으로 대체 (추가 배율 적용)
#
# bending    : 플랜지 전체 고정, 바닥 중앙부에 하향 균등 분포 하중
# bending-xspan: Y방향 양단 고정, X스팬 굽힘
# bending-yspan: X방향 양단 고정, Y스팬 굽힘
# twisting   : 대각 두 코너 고정, 반대 두 코너에 ±하중 (비틀림)
# twisting-alt: 반전 대각 비틀림
# lifting    : 네 코너에 상향 하중 (리프팅)
--w-bending       1.0  -10.0
--w-bending-xspan 1.0  -10.0
--w-bending-yspan 1.0  -10.0
--w-twisting      1.0  -10.0
--w-twisting-alt  1.0  -10.0
--w-lifting       1.0   10.0
--w-basic-weight  1.0

# ── 출력 및 시각화 ───────────────────────────────────────────────────────────
# 결과 파일 저장 경로와 시각화 옵션입니다.
# export   : 최적화 결과 비드 형상을 LS-DYNA .k 형식으로 저장할 경로
# mesh-size: 경계 조건 노드 탐색 기준 메시 크기 [mm]. mesh-z 와 동일하게 설정 권장
# no-gui   : 최적화 중 모니터링 GUI 비활성화 (헤드리스 서버 환경에서 사용)
# no-viz   : 최적화 완료 후 3D 결과 시각화 생략
--export        industrial_bead.k
--mesh-size     10.0
# --no-gui
# --no-viz
